# Binary Search - Iterative and Recursive

# Notes and Step-by-Step Explanation:
# - Binary search is an efficient algorithm to find an element in a **sorted** list.
# - It works by repeatedly dividing the search interval in half.
# - If the value of the search key is less than the item in the middle, narrow the interval to the lower half.
# - Otherwise, narrow it to the upper half.
# - Repeat until the value is found or the interval is empty.

# Iterative Approach
def binary_search_iterative(arr, target):
    """
    Performs binary search iteratively.

    Step-by-step:
    1. Set low = 0 and high = len(arr) - 1 (start and end of the list).
    2. While low <= high:
        a. Find the middle index: mid = (low + high) // 2.
        b. If arr[mid] == target, return mid (target found).
        c. If arr[mid] < target, search the right half (set low = mid + 1).
        d. If arr[mid] > target, search the left half (set high = mid - 1).
    3. If the loop ends, return -1 (target not found).

    Args:
        arr (list): Sorted list to search in.
        target: Element to search for.

    Returns:
        int: Index of the target if found, else -1.
    """
    low = 0
    high = len(arr) - 1

    while low <= high:
        mid = (low + high) // 2  # Step 2a

        if arr[mid] == target:    # Step 2b
            return mid
        elif arr[mid] < target:   # Step 2c
            low = mid + 1
        else:                    # Step 2d
            high = mid - 1

    return -1  # Step 3: Target not found


# Recursive Approach
def binary_search_recursive(arr, target, low, high):
    """
    Performs binary search recursively.

    Step-by-step:
    1. If low > high, return -1 (base case: target not found).
    2. Find the middle index: mid = (low + high) // 2.
    3. If arr[mid] == target, return mid (target found).
    4. If arr[mid] < target, search right half: call recursively with low = mid + 1.
    5. If arr[mid] > target, search left half: call recursively with high = mid - 1.

    Args:
        arr (list): Sorted list to search in.
        target: Element to search for.
        low (int): Starting index of the search space.
        high (int): Ending index of the search space.

    Returns:
        int: Index of the target if found, else -1.
    """
    if low > high:  # Step 1
        return -1

    mid = (low + high) // 2  # Step 2

    if arr[mid] == target:   # Step 3
        return mid
    elif arr[mid] < target:  # Step 4
        return binary_search_recursive(arr, target, mid + 1, high)
    else:                    # Step 5
        return binary_search_recursive(arr, target, low, mid - 1)


# Example Usage:
if __name__ == "__main__":
    arr = [2, 5, 7, 8, 11, 12]
    target = 13

    # Iterative Example
    index_iterative = binary_search_iterative(arr, target)
    if index_iterative != -1:
        print(f"Iterative: Element found at index {index_iterative}")
    else:
        print(f"Iterative: Element not found")

    # Recursive Example
    index_recursive = binary_search_recursive(arr, target, 0, len(arr) - 1)
    if index_recursive != -1:
        print(f"Recursive: Element found at index {index_recursive}")
    else:
        print(f"Recursive: Element not found")

# Summary:
# - Both approaches repeatedly divide the search space in half.
# - Iterative uses a loop; recursive uses function calls.
# - Both return the index if found, or -1 if not found.
